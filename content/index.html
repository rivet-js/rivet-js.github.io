Current milestone

A reactive counter, without compiler magic.

The public API already includes state(), computed(), effect(), html, and render().

import { state, html } from "rivet";

export function Counter() {
    const count = state(0);

    return html`
        <button onclick=${() => count.value++}>
            Count: ${count}
        </button>
    `;
}
