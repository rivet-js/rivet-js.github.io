import { html, key, mount, state } from "rivet";

export function DynamicList() {
    const visible = state(true);
    const rows = state([
        { id: 1, label: state("HTML") },
        { id: 2, label: state("JavaScript") },
        { id: 3, label: state("Rivet") }
    ]);

    const reverse = () => {
        rows.value = [...rows.value].reverse();
    };

    return html`
        <section>
            <button type="button" onclick=${() => { visible.value = !visible.value; }}>
                ${() => visible.value ? "Hide" : "Show"} list
            </button>
            <button type="button" onclick=${reverse}>Reverse</button>
            ${() => visible.value
                ? html`<ul>${rows.value.map((row) => key(row.id, html`<li>${row.label}</li>`))}</ul>`
                : html`<p>List hidden.</p>`}
        </section>
    `;
}

mount("#app", DynamicList());
